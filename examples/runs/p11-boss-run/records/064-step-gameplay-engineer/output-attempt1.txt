"""D-022 probes at both rates; controlled fixtures are not reachability evidence."""

import contextlib
import io
import re
import unittest

import play
from game import Shot
from run_driver import drive, read_script
from session import RunSession


class SupplementsTest(unittest.TestCase):
    def output(self, script, view="player"):
        setup, inputs, rules = play.read_script(script)
        stream = io.StringIO()
        with contextlib.redirect_stdout(stream):
            result = play.play(setup, inputs, view, rules)
        return stream.getvalue(), result

    def test_legacy_note_scope(self):
        note = 'note: legacy route — boss status and messages are incomplete; add "mode room" for the full boss display'
        for fps in (30, 60):
            for room in ("1", "2", "3", "4", "5", "empty"):
                for view in ("player", "trace"):
                    text, _ = self.output(f"room {room}\nfps {fps}\nend 0.1\n", view)
                    self.assertEqual(text.count(note), int(room == "5" and view == "player"))
                    if room == "5" and view == "player":
                        self.assertEqual(text.splitlines()[0], note)

    def test_public_boss_display_and_no_ids(self):
        for fps in (30, 60):
            text, _ = self.output(f"mode room\nroom 5\nfps {fps}\nlimit 8\n")
            for fragment in ("boss B — HP", "next: charge", "the boss prepares a charge"):
                self.assertIn(fragment, text)
            self.assertIsNone(re.search(r"\bS\d+\b", text))
            self.assertEqual(text.count("summary:"), 1)

    def test_shot_reasons_and_recovery(self):
        for fps in (30, 60):
            for reason in ("lifetime", "edge", "hit", "dash", "protection"):
                with self.subTest(fps=fps, reason=reason):
                    s = RunSession(fps)
                    s._enter("5", 3)
                    g, boss = s.game, s.game.enemies[0]
                    boss.x, boss.y = (0.05 if reason == "edge" else 10), 4
                    boss.phase, boss.attack_name = "warning", "short shot"
                    boss.phase_left, boss.aim = g.dt, (-1, 0)
                    g.place_player(0, 0)
                    s.step()
                    fired = s.t
                    if reason in ("hit", "dash", "protection"):
                        g.place_player(9.7, 4)
                    if reason == "dash":
                        s.press("dash", "left")
                    elif reason == "protection":
                        g.hurt_left, g.invulnerable = 0.8, True
                    for _ in range(fps):
                        s.step()
                    emitted = [e for e in s.events if e["kind"] == "shot_fired"]
                    removed = [e for e in s.events if e["kind"] == "shot_removed"]
                    self.assertEqual(len(emitted), 1)
                    self.assertEqual(len(removed), 1)
                    self.assertEqual(emitted[0]["details"], {"id": "S1", "source": "boss"})
                    expected = "absorbed" if reason in ("dash", "protection") else reason
                    self.assertEqual(removed[0]["details"]["reason"], expected)
                    self.assertEqual(removed[0]["details"]["id"], "S1")
                    self.assertEqual(g.shots, [])
                    recovery = [e for e in s.events if e["kind"] == "boss_recovery"]
                    self.assertEqual(len(recovery), 1)
                    self.assertAlmostEqual(recovery[0]["run_time"] - fired, 1)

    def test_clear_and_retry_ids(self):
        for fps in (30, 60):
            s = RunSession(fps)
            g = s.game
            shot = Shot(0, 0, 0, 0)
            g._emit_shot(shot, "spitter1")
            for enemy in g.enemies:
                enemy.hp = 0
            s.step()
            removed = [e for e in s.events if e["kind"] == "shot_removed"]
            self.assertEqual([e["details"]["reason"] for e in removed], ["cleared"])
            for _ in range(fps):
                s.step()
            s.game._emit_shot(Shot(0, 0, 0, 0), "spitter1")
            self.assertEqual(s.game.shots[0].id, "S2")
            s.game.hearts = 0
            s.step()
            self.assertTrue(s.control("retry"))
            s.game._emit_shot(Shot(0, 0, 0, 0), "spitter1")
            self.assertEqual(s.game.shots[0].id, "S1")

    def test_interval_validation(self):
        for key in ("view-every", "trace-every"):
            for value in ("0.49", "60.1", "nan", "inf", "no"):
                with self.assertRaisesRegex(ValueError, "interval"):
                    read_script(f"mode run\n{key} {value}\n")
        self.assertTrue(read_script("trace-every frame\n")[0]["trace_every"])

    def test_intervals_preserve_events_and_results(self):
        for fps in (30, 60):
            results = []
            messages = []
            for interval in (0.5, 10):
                text, (s, measurement) = self.output(
                    f"mode run\nfps {fps}\nlimit 10\nview-every {interval}\n")
                results.append((s.state, s.t, s.game.hearts, s.events))
                messages.append([line for line in text.splitlines() if line.startswith("[attempt")])
                self.assertEqual(measurement["attempt_count"], 1)
                row = measurement["attempts"][0]
                self.assertEqual(row["run_seconds"], s.t)
                self.assertEqual(row["hits"], sum(e["kind"] == "hurt" for e in s.events))
                self.assertEqual(row["attacks"], sum(e["kind"] == "attack" for e in s.events))
            self.assertEqual(results[0], results[1])
            self.assertEqual(messages[0], messages[1])

    def test_twenty_minute_periodic_bounds(self):
        # Normal-run fixture: freeze threats without clearing rooms, to exercise the full limit.
        from run_driver import Output
        for fps in (30, 60):
            for view, interval, maximum in (("player", 10, 121), ("trace", 5, 241)):
                setup, inputs, rules = read_script(
                    f"mode run\nfps {fps}\nlimit 1200\ndriver-limit 1201\n"
                    "view-every 10\ntrace-every 5\n")
                output = None

                def observe(s, force):
                    nonlocal output
                    if output is None:
                        for enemy in s.game.enemies:
                            enemy.x, enemy.y = 0, 0
                            enemy.phase, enemy.phase_left = "recovery", 2000
                        output = Output(s, setup, view)
                    output.emit(force)

                stream = io.StringIO()
                with contextlib.redirect_stdout(stream):
                    s, measurement = drive(setup, inputs, rules, observe)
                self.assertEqual(s.t, 1200)
                self.assertEqual(measurement["attempts"][0]["outcome"], "unfinished")
                lines = stream.getvalue().splitlines()
                if view == "player":
                    # Initial and periodic pictures; the final limit has no player message.
                    count = sum(line.startswith("run — room") for line in lines)
                else:
                    import ast
                    states = [ast.literal_eval(line) for line in lines]
                    count = sum(not state["events"] for state in states) + 1
                self.assertLessEqual(count, maximum)
