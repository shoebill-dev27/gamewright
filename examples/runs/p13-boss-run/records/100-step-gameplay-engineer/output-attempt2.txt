        """Controlled contact fixture; diagnostic reads occur after combat."""
        from test_g002_evidence import emit_rows

        rows = []
        for fps in (30, 60):
            game = Game(fps=fps, room="5")

            boss.phase, boss.attack_name = "active", "charge"
            boss.aim, boss.phase_left = (-1.0, 0.0), 0.3
            first = len(game.events)

            self.assertFalse(boss.spent)

                game.x, game.y, game.hearts, tuple(game.events),
