"""G-003: dash availability display. Display only; the dash rules are unchanged."""

import contextlib
import copy
import io
import sys
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
import play  # noqa: E402
from game import Game, dash_seconds  # noqa: E402


def would_accept(game):
    probe = copy.deepcopy(game)
    before = len(probe.events)
    probe.press("dash", "right")
    return not any(k == "ignored" for _, k, _ in probe.events[before:])


def times(game, kind):
    return [t for t, k, _ in game.events if k == kind]


def run_until(game, t):
    while game.t < t - 1e-9:
        game.step()


def capture(script):
    setup, inputs, rules = play.read_script(script)
    out = io.StringIO()
    with contextlib.redirect_stdout(out):
        play.play(setup, inputs, "player", rules)
    return out.getvalue()


class DashDisplayTest(unittest.TestCase):
    def test_status_matches_what_a_press_does_every_frame(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="empty")
            game.hearts = 50
            hit = False
            for _ in range(4 * fps):
                if abs(game.t - 0.5) < 1e-9:
                    game.press("dash", "right")
                if game.t >= 1.3 - 1e-9 and not hit:
                    hit = True
                    game._hurt("grub")
                game.step()
                self.assertEqual(game.dash_status()[0] == "ready", would_accept(game), (fps, game.t))

    def test_ready_comes_once_at_dash_end_plus_0_6(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="empty")
            game.press("dash", "right")
            run_until(game, 3.0)
            start, end, ready = times(game, "dash")[0], times(game, "dash_end")[0], times(game, "dash_ready")
            self.assertEqual(len(ready), 1, fps)
            self.assertAlmostEqual(ready[0] - end, 0.6, delta=1.0 / fps + 1e-9)
            self.assertAlmostEqual(ready[0] - start, 0.85, delta=1.0 / fps + 1e-9)

    def test_no_ready_message_without_a_dash(self):
        game = Game(fps=60, room="empty")
        game.spawn("grub", 8.5, 4)
        run_until(game, 3.0)
        self.assertTrue(times(game, "hurt"))
        self.assertEqual(times(game, "dash_ready"), [])

    def test_hit_during_cooldown_does_not_change_the_cooldown(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="empty")
            game.hearts = 50
            game.press("dash", "right")
            while not times(game, "dash_end"):
                game.step()
            end = times(game, "dash_end")[0]
            run_until(game, end + 0.1)
            game._hurt("grub")
            run_until(game, 3.0)
            ready = times(game, "dash_ready")
            self.assertEqual(len(ready), 1, fps)
            self.assertAlmostEqual(ready[0] - end, 0.6, delta=1.0 / fps + 1e-9)

    def test_stun_that_outlasts_the_cooldown_delays_ready(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="empty")
            game.hearts = 50
            game.press("dash", "right")
            while not times(game, "dash_end"):
                game.step()
            end = times(game, "dash_end")[0]
            run_until(game, end + 0.5)
            hit_at = game.t
            game._hurt("grub")
            run_until(game, 3.0)
            ready = times(game, "dash_ready")
            self.assertEqual(len(ready), 1, fps)
            self.assertAlmostEqual(ready[0] - hit_at, 0.3, delta=1.0 / fps + 1e-9)

    def test_attack_recovery_does_not_block_a_dash(self):
        game = Game(room="empty")
        game.press("attack")
        self.assertGreater(game.recovery, 0)
        game.press("dash", "right")
        self.assertEqual(len(times(game, "dash")), 1)
        self.assertEqual(game.dash_status()[0], "active")

    def test_room_entry_starts_ready(self):
        self.assertEqual(Game(room="1").dash_status(), ("ready", 0.0))

    def test_texts(self):
        d = play.Player.describe
        self.assertEqual(d("dash", {"direction": "right", "usable": 0.85}), "you dash right — usable again in 0.9 s")
        self.assertEqual(
            d("ignored", {"action": "dash", "reason": "dash not ready", "left": 0.5, "why": None}),
            "dash not ready — 0.5 s left",
        )
        self.assertEqual(
            d("ignored", {"action": "dash", "reason": "stunned", "left": 0.2, "why": "hit-stun"}),
            "dash not ready — 0.2 s left (hit-stun)",
        )
        self.assertEqual(d("ignored", {"action": "attack", "reason": "recovering"}), "(nothing happens)")
        self.assertIsNone(d("ignored", {"action": "dash", "left": 0.5, "quiet": True}))
        self.assertEqual(d("dash_ready", {}), "dash ready")
        self.assertEqual(dash_seconds(0.45), "0.5")
        self.assertEqual(dash_seconds(0.6), "0.6")
        self.assertEqual(dash_seconds(0.85), "0.9")
        self.assertEqual(dash_seconds(1e-9), "0.1")

    def test_legacy_player_view(self):
        for fps in (30, 60):
            out = capture(
                f"room empty\nfps {fps}\n0.6 dash right\n0.7 dash right\n1.0 dash right\n1.4 dash right\nend 3\n"
            )
            self.assertIn("hearts 3/3 — dash: ready", out, fps)
            self.assertIn("you dash right — usable again in 0.9 s", out, fps)
            self.assertEqual(out.count("dash not ready"), 3, fps)
            self.assertIn("(dashing)", out, fps)
            self.assertIn("dash: 0.5 s", out, fps)
            self.assertIn("dash: active", out, fps)
            self.assertEqual(len([l for l in out.splitlines() if l.strip().endswith("dash ready")]), 1, fps)

    def test_text_rule_sees_dash_ready(self):
        out = capture(
            'room empty\n0.1 dash right\nwhen text "dash ready" then dash left\nend 2\n'
        )
        self.assertEqual(out.count("you dash right"), 1)
        self.assertEqual(out.count("you dash left"), 1)

    def test_public_driver_status_line(self):
        setup, inputs, rules = play.read_script(
            "mode room\nroom empty\nfps 60\nlimit 2\ndriver-limit 3\n0.6 dash right\n"
        )
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            play.play(setup, inputs, "player", rules)
        text = out.getvalue()
        self.assertIn("— combat — dash: ready", text)
        self.assertIn("— combat — dash: active", text)
        self.assertIn("usable again in 0.9 s", text)
        self.assertEqual(len([l for l in text.splitlines() if l.endswith("] dash ready")]), 1)
