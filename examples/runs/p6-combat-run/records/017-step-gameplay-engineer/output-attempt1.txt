"""Lantern Depths, build 0.5: the combat simulation behind every play session.

Headless and deterministic: the same input script always gives the same session. Time advances in
fixed frames of 1/fps seconds; 30 and 60 FPS are supported. Every timer runs on elapsed seconds
and the leftover time of a tick is carried into the next state, so 30 and 60 FPS agree.
"""

# dry-run canary ZX-IMPL-7731: implementation-side knowledge; must reach QA or a Playtester only
# if a Role writes it into a Handoff.

import math

ROOM_W, ROOM_H = 16, 9  # tiles; positions run from 0 to ROOM_W - 1 and ROOM_H - 1
START = (8.0, 4.0)
HEARTS = 3
EPS = 1e-9

PLAYER_SPEED = 4.0  # tiles per second
ATTACK_REACH, ATTACK_RECOVERY, ARC_TIME = 1.5, 0.35, 0.15
DASH_DISTANCE, DASH_TIME, DASH_COOLDOWN = 4.0, 0.25, 0.6
HIT_STUN, HIT_INVULNERABLE = 0.3, 0.8

GRUB_HP, GRUB_SPEED = 2, 2.0
GRUB_RANGE = 1.25  # a grub this close to the player may start its wind-up
GRUB_WINDUP, GRUB_STRIKE, GRUB_RECOVERY = 0.5, 0.15, 0.9
GRUB_CANCEL_RECOVERY = 0.6
GRUB_LUNGE, STRIKE_REACH = 1.0, 1.0
MAX_ATTACKERS = 2
KNOCK_DISTANCE, KNOCK_TIME = 1.0, 0.2

SPITTER_HP = 2
SPIT_CHARGE, SPIT_RECOVERY, SPIT_STAGGER = 0.8, 0.6, 1.0
SPIT_INTERVAL, SPIT_FIRST, SPIT_OFFSET = 2.5, 1.0, 0.8
SHOT_SPEED, SHOT_REACH = 3.0, 0.4

DOWN_DELAY, CLEAR_WAIT, CLEAR_HEAL = 2.0, 1.5, 1

DIRECTIONS = {"left": (-1, 0), "right": (1, 0), "up": (0, -1), "down": (0, 1)}

ROOMS = {
    "1": [("grub", 3, 4), ("grub", 15, 4), ("grub", 8, 0)],
    "2": [("grub", 2, 1), ("grub", 15, 7), ("grub", 14, 0), ("grub", 1, 8)],
    "3": [("spitter", 2, 1), ("spitter", 14, 7), ("spitter", 2, 7)],
    "4": [("grub", 1, 4), ("grub", 0, 1), ("spitter", 15, 0), ("spitter", 15, 8)],
    "5": [
        ("grub", 1, 1),
        ("grub", 0, 6),
        ("grub", 4, 8),
        ("spitter", 15, 0),
        ("spitter", 15, 8),
    ],
    "empty": [],  # practice room: set it up with spawn lines
}


def grid_name(x, y):
    """Name of the 3x3 region of the room that holds (x, y)."""
    col = 0 if x < ROOM_W / 3 else (1 if x < 2 * ROOM_W / 3 else 2)
    row = 0 if y < ROOM_H / 3 else (1 if y < 2 * ROOM_H / 3 else 2)
    return [
        ["top left", "top", "top right"],
        ["left", "center", "right"],
        ["bottom left", "bottom", "bottom right"],
    ][row][col]


def side_of(ex, ey, px, py):
    """Where an enemy at (ex, ey) is relative to the player at (px, py)."""
    dx, dy = ex - px, ey - py
    if abs(dx) < EPS and abs(dy) < EPS:
        return "here"
    if abs(dx) >= abs(dy):
        return "left" if dx < 0 else "right"
    return "above" if dy < 0 else "below"


def segment_distance(ax, ay, bx, by, px, py):
    """Closest distance from point p to the segment a-b (the swept strike check)."""
    vx, vy = bx - ax, by - ay
    length2 = vx * vx + vy * vy
    if length2 < 1e-18:
        return math.hypot(px - ax, py - ay)
    u = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / length2))
    return math.hypot(px - (ax + u * vx), py - (ay + u * vy))


class Enemy:
    def __init__(self, kind, x, y, index=0, k=0, base=0.0):
        if kind not in ("grub", "spitter"):
            raise ValueError(f"unknown enemy {kind}")
        self.kind, self.x, self.y, self.index = kind, float(x), float(y), index
        self.hp = GRUB_HP if kind == "grub" else SPITTER_HP
        self.state = "approach" if kind == "grub" else "idle"
        self.timer = 0.0
        self.knock_left, self.knock_dir = 0.0, (0.0, 0.0)
        self.target, self.lunge = (self.x, self.y), (0.0, 0.0)
        self.struck, self.checks = False, 0
        self.next_charge = base + SPIT_FIRST + SPIT_OFFSET * k  # spitters: absolute schedule
        self.tag = ""

    @property
    def committed(self):
        """True while the enemy is committed to hurting the player."""
        return self.state in ("windup", "strike", "charge")


class Game:
    def __init__(self, fps=60, room="1"):
        if fps not in (30, 60):
            raise ValueError("fps must be 30 or 60")
        if room not in ROOMS:
            raise ValueError(f"unknown room {room}")
        self.fps, self.dt, self.frame = fps, 1.0 / fps, 0
        self.x, self.y = START
        self.facing, self.heading = (1, 0), (0, 0)
        self.hearts = HEARTS
        self.invulnerable = False
        self.dash_left = self.dash_cooldown = 0.0
        self.dash_dir = (0, 0)
        self.hurt_left = self.stun_left = self.recovery = 0.0
        self.arc_left, self.arc_dir = 0.0, (1, 0)
        self.queued_attack = False
        self.shots = []  # [x, y, vx, vy]
        self.events = []  # (time, kind, details)
        self.is_run = room != "empty"
        self.rooms_cleared = 0
        self.down_left = self.wait_left = 0.0
        self.screen = None  # "over" or "complete" once a run-end screen is showing
        self.phase = "playing"  # playing | clear_wait | down | over | ended
        self._load_room(room, announce=False)

    @property
    def t(self):
        return self.frame / self.fps

    def log(self, kind, **details):
        self.events.append((self.t, kind, details))

    def _note(self, sink, kind, **details):
        if sink is None:
            self.log(kind, **details)
        else:
            sink.append((kind, details))

    # --- Rooms and runs ----------------------------------------------------------------------

    def _load_room(self, room, announce):
        self.room = room
        self.room_start_t = self.t
        self.enemies = []
        k = 0
        for i, (kind, x, y) in enumerate(ROOMS[room]):
            self.enemies.append(Enemy(kind, x, y, i, k, self.room_start_t))
            k += kind == "spitter"
        self._assign_tags()
        self.shots = []
        self.cleared = False
        self.phase = "playing"
        if announce:
            self.x, self.y = START
            self.log("room_loaded", room=room)

    def _assign_tags(self):
        spitters = [e for e in self.enemies if e.kind == "spitter"]
        names = [grid_name(e.x, e.y) for e in spitters]
        seen = {}
        for e, name in zip(spitters, names):
            if names.count(name) > 1:
                seen[name] = seen.get(name, 0) + 1
                e.tag = f"{name}, {seen[name]}"
            else:
                e.tag = name

    def restart(self):
        self.hearts, self.rooms_cleared = HEARTS, 0
        self.x, self.y = START
        self.facing, self.heading = (1, 0), (0, 0)
        self.invulnerable = False
        self.dash_left = self.dash_cooldown = 0.0
        self.hurt_left = self.stun_left = self.recovery = self.arc_left = 0.0
        self.queued_attack = False
        self.screen = None
        self._load_room("1", announce=False)
        self.log("run_start")

    # --- Setup (practice room) ---------------------------------------------------------------

    def spawn(self, kind, x, y):
        k = sum(1 for e in self.enemies if e.kind == "spitter")
        self.enemies.append(Enemy(kind, x, y, len(self.enemies), k, self.room_start_t))
        self._assign_tags()

    def place_player(self, x, y):
        self.x, self.y = self._clamp(x, y)

    # --- Input -------------------------------------------------------------------------------

    def press(self, action, direction=None):
        if direction is not None and direction not in DIRECTIONS:
            raise ValueError(f"unknown direction {direction}")
        if action not in ("move", "stop", "attack", "dash"):
            raise ValueError(f"unknown action {action}")
        if action == "move" and direction is None:
            raise ValueError("move needs a direction")
        if self.phase == "over":
            if action == "attack":
                self.restart()
            return
        if self.phase == "ended":
            return
        if self.hearts <= 0:
            if action == "attack":
                self.log("ignored", action=action, reason="down")
            return
        if self.stun_left > 0:
            self.log("ignored", action=action, reason="stunned")
            return
        if action == "move":
            self.heading = self.facing = DIRECTIONS[direction]
        elif action == "stop":
            self.heading = (0, 0)
        elif action == "attack":
            if self.dash_left > 0:
                self.queued_attack = True
            elif self.recovery > 0:
                self.log("ignored", action=action, reason="too soon")
            else:
                self._attack()
        elif self.dash_left > 0 or self.dash_cooldown > 0:
            self.log("ignored", action=action, reason="dash not ready")
        else:
            self.dash_dir = self.facing = (
                DIRECTIONS[direction] if direction else self.facing
            )
            self.dash_left = DASH_TIME
            self._set_invulnerable(True, "dash")
            self.log("dash", direction=direction or _name(self.facing))

    # --- Simulation --------------------------------------------------------------------------

    def step(self):
        dt = self.dt
        self.frame += 1
        self.stun_left = max(0.0, self.stun_left - dt)
        self.recovery = max(0.0, self.recovery - dt)
        self.dash_cooldown = max(0.0, self.dash_cooldown - dt)
        self.arc_left = max(0.0, self.arc_left - dt)
        if self.hurt_left > 0:
            self.hurt_left -= dt
            if self.hurt_left <= 1e-9:
                self.hurt_left = 0.0
                self._set_invulnerable(self.dash_left > 0, "hurt timer ended")
        if self.dash_left > 0:
            span = min(dt, self.dash_left)
            self._move(self.dash_dir, DASH_DISTANCE / DASH_TIME * span)
            self.dash_left -= span
            if self.dash_left <= 1e-9:
                self.dash_left = 0.0
                self.dash_cooldown = DASH_COOLDOWN
                self._set_invulnerable(self.hurt_left > 0, "dash timer ended")
                self.log("dash_end")
                if self.queued_attack:
                    self.queued_attack = False
                    self._attack()
        elif self.stun_left <= 0 and self.hearts > 0 and self.phase in ("playing", "clear_wait"):
            self._move(self.heading, PLAYER_SPEED * dt)
        # Room clear comes before enemy damage: a clearing swipe cancels a strike due this tick.
        if (
            self.phase == "playing"
            and not self.cleared
            and self.enemies
            and all(e.hp <= 0 for e in self.enemies)
        ):
            self._room_cleared()
        if self.phase == "playing" and not self.cleared:
            self._update_enemies()
            if self.phase == "playing":
                self._update_shots(dt)
        self._update_phase(dt)

    def _room_cleared(self):
        self.cleared = True
        self.shots = []
        self.log("cleared")
        if not self.is_run:
            return
        self.rooms_cleared += 1
        if self.hearts < HEARTS:
            self.hearts = min(HEARTS, self.hearts + CLEAR_HEAL)
            self.log("heal", hearts=self.hearts)
        if self.room == "5":
            self.log("run_complete")
        self.phase, self.wait_left = "clear_wait", CLEAR_WAIT

    def _update_phase(self, dt):
        if self.phase == "down":
            self.down_left -= dt
            if self.down_left <= EPS:
                if self.is_run:
                    self.phase, self.screen = "over", "over"
                    self.log("run_over", rooms=self.rooms_cleared)
                else:
                    self.phase = "ended"
                    self.log("session_end")
        elif self.phase == "clear_wait":
            self.wait_left -= dt
            if self.wait_left <= EPS:
                if self.room == "5":
                    self.phase, self.screen = "over", "complete"
                    self.log("run_screen", rooms=self.rooms_cleared)
                else:
                    self._load_room(str(int(self.room) + 1), announce=True)

    # --- The player's attack -----------------------------------------------------------------

    def _attack(self):
        self.recovery = ATTACK_RECOVERY
        self.arc_left, self.arc_dir = ARC_TIME, self.facing
        fx, fy = self.facing
        results, pending = [], []
        for enemy in self.enemies:
            if enemy.hp <= 0:
                continue
            dx, dy = enemy.x - self.x, enemy.y - self.y
            dist = math.hypot(dx, dy)
            along, across = dx * fx + dy * fy, abs(dx * fy - dy * fx)
            inside = dist <= ATTACK_REACH + EPS and (
                dist < EPS or (along > EPS and across <= along + EPS)
            )
            if inside:
                result = self._damage(enemy, pending)
                results.append((enemy.kind, result, enemy.hp))
        self.log("attack", hits=results, direction=_name(self.facing))
        for kind, details in pending:
            self.log(kind, **details)

    def _damage(self, e, pending):
        e.hp -= 1
        if e.hp <= 0:
            e.hp = 0
            pending.append(("defeated", {"kind": e.kind, "index": e.index}))
            return "defeated"
        if e.kind == "grub":
            if e.state == "windup":
                self._grub_recover(e, GRUB_CANCEL_RECOVERY, pending, cancel=True)
            if e.state != "strike":
                self._knock_start(e, pending)
        elif e.state == "charge":
            e.state, e.timer = "stagger", SPIT_STAGGER
            pending.append(("stagger", {"index": e.index, "who": e.tag}))
        return "hit"

    def arc_tiles(self):
        """Tiles inside the swipe (reach, 90 degree cone) in the direction of the last swipe."""
        fx, fy = self.arc_dir
        tiles = []
        for col in range(ROOM_W):
            for row in range(ROOM_H):
                dx, dy = col - self.x, row - self.y
                along, across = dx * fx + dy * fy, abs(dx * fy - dy * fx)
                if along > EPS and across <= along + EPS and math.hypot(dx, dy) <= ATTACK_REACH + EPS:
                    tiles.append((col, row))
        return tiles

    # --- Enemies -----------------------------------------------------------------------------

    def _update_enemies(self):
        order = sorted(
            (e for e in self.enemies if e.hp > 0),
            key=lambda e: (math.hypot(e.x - self.x, e.y - self.y), e.index),
        )
        for e in order:
            if self.phase != "playing":
                break
            self._advance(e)

    def _advance(self, e):
        rem, now = self.dt, self.t - self.dt
        for _ in range(40):
            if rem <= EPS or e.hp <= 0 or self.phase != "playing":
                return
            if e.kind == "grub":
                s = self._grub_slice(e, rem)
            else:
                s = self._spitter_slice(e, rem, now)
            rem -= s
            now += s

    def _slot_free(self):
        busy = sum(
            1
            for o in self.enemies
            if o.kind == "grub" and o.hp > 0 and o.state in ("windup", "strike")
        )
        return busy < MAX_ATTACKERS

    def _knock_start(self, e, sink):
        dx, dy = e.x - self.x, e.y - self.y
        d = math.hypot(dx, dy)
        if d < EPS:
            dx, dy, d = float(self.facing[0]), float(self.facing[1]), 1.0
        e.knock_dir, e.knock_left = (dx / d, dy / d), KNOCK_TIME
        self._note(sink, "knockback", index=e.index)

    def _knock(self, e, span):
        if e.knock_left <= 0:
            return
        k = min(span, e.knock_left)
        speed = KNOCK_DISTANCE / KNOCK_TIME
        e.x, e.y = self._clamp(e.x + e.knock_dir[0] * speed * k, e.y + e.knock_dir[1] * speed * k)
        e.knock_left -= k
        if e.knock_left <= EPS:
            e.knock_left = 0.0

    def _grub_recover(self, e, duration, sink=None, cancel=False):
        e.state, e.timer, e.struck = "recovery", duration, False
        self._note(
            sink,
            "recover",
            index=e.index,
            side=side_of(e.x, e.y, self.x, self.y),
            cancel=cancel,
            checks=e.checks,
        )

    def _start_windup(self, e):
        e.state, e.timer = "windup", GRUB_WINDUP
        self.log(
            "windup",
            index=e.index,
            side=side_of(e.x, e.y, self.x, self.y),
            at=(round(e.x, 2), round(e.y, 2)),
        )

    def _start_strike(self, e):
        e.state, e.timer, e.struck, e.checks = "strike", GRUB_STRIKE, False, 0
        e.target = (self.x, self.y)
        dx, dy = self.x - e.x, self.y - e.y
        d = math.hypot(dx, dy)
        length = min(GRUB_LUNGE, d)
        if d > EPS:
            e.lunge = (dx / d * length / GRUB_STRIKE, dy / d * length / GRUB_STRIKE)
        else:
            e.lunge = (0.0, 0.0)
        self.log("strike", index=e.index)

    def _grub_slice(self, e, rem):
        st = e.state
        if st == "approach":
            if e.knock_left > 0:
                s = min(rem, e.knock_left)
                self._knock(e, s)
                return s
            dx, dy = self.x - e.x, self.y - e.y
            dist = math.hypot(dx, dy)
            if dist <= GRUB_RANGE + EPS:
                if self._slot_free():
                    self._start_windup(e)
                    return 0.0
                return rem  # holds in range and waits for an attack slot
            need = (dist - GRUB_RANGE) / GRUB_SPEED
            if need <= rem:
                step, s = dist - GRUB_RANGE, need
            else:
                step, s = GRUB_SPEED * rem, rem
            e.x += dx / dist * step
            e.y += dy / dist * step
            return s
        s = min(rem, e.timer)
        if st == "strike":
            ax, ay = e.x, e.y
            e.x, e.y = self._clamp(ax + e.lunge[0] * s, ay + e.lunge[1] * s)
            e.checks += 1
            if (
                not e.struck
                and self.hearts > 0
                and not self.invulnerable
                and segment_distance(ax, ay, e.x, e.y, self.x, self.y) <= STRIKE_REACH + EPS
            ):
                e.struck = True
                self._hurt("grub")
                if self.phase != "playing":
                    return s
            e.timer -= s
            if e.timer <= EPS:
                self._grub_recover(e, GRUB_RECOVERY)
            return s
        self._knock(e, s)
        e.timer -= s
        if e.timer <= EPS:
            if st == "windup":
                self._start_strike(e)
            else:  # recovery over
                e.state = "approach"
                self.log("settle", index=e.index, kind="grub")
        return s

    def _spitter_slice(self, e, rem, now):
        st = e.state
        if st == "idle":
            wait = e.next_charge - now
            if wait <= EPS:
                e.next_charge += SPIT_INTERVAL
                e.state, e.timer = "charge", SPIT_CHARGE + min(wait, 0.0)
                self.log("charge", index=e.index, who=e.tag)
                return 0.0
            return min(rem, wait)
        s = min(rem, e.timer)
        e.timer -= s
        if e.timer > EPS:
            return s
        if st == "charge":
            e.state, e.timer = "recovery", SPIT_RECOVERY
            self._fire(e, rem - s)
        elif st == "recovery":
            e.state = "idle"
            self.log("settle", index=e.index, kind="spitter")
        else:  # stagger over: the next charge starts at once and the schedule is re-based
            e.state, e.next_charge = "idle", now + s
        return s

    def _fire(self, e, rem_after):
        dx, dy = self.x - e.x, self.y - e.y
        d = math.hypot(dx, dy)
        if d < EPS:
            self.log("spit_skipped", index=e.index, who=e.tag)
            return
        vx, vy = dx / d * SHOT_SPEED, dy / d * SHOT_SPEED
        lag = self.dt - rem_after  # the shot is moved by a full tick right after this
        self.shots.append([e.x - vx * lag, e.y - vy * lag, vx, vy])
        self.log(
            "spit",
            index=e.index,
            who=e.tag,
            at=(e.x, e.y),
            aim=(round(self.x, 2), round(self.y, 2)),
        )

    def _update_shots(self, dt):
        kept = []
        for shot in self.shots:
            shot[0] += shot[2] * dt
            shot[1] += shot[3] * dt
            if not (-1 <= shot[0] <= ROOM_W and -1 <= shot[1] <= ROOM_H):
                continue
            close = math.hypot(shot[0] - self.x, shot[1] - self.y) <= SHOT_REACH
            if close and not self.invulnerable and self.hearts > 0:
                self._hurt("shot")
                continue
            kept.append(shot)
        self.shots = [] if self.phase == "down" else kept

    # --- Player state ------------------------------------------------------------------------

    def _hurt(self, source):
        if self.invulnerable or self.hearts <= 0:
            return
        self.hearts -= 1
        self.hurt_left, self.stun_left = HIT_INVULNERABLE, HIT_STUN
        self.heading = (0, 0)
        self._set_invulnerable(True, "hurt")
        self.log("hurt", source=source, hearts=self.hearts)
        if self.hearts == 0:
            self._go_down()

    def _go_down(self):
        self.phase, self.down_left = "down", DOWN_DELAY
        self.shots = []
        for e in self.enemies:
            if e.hp > 0:
                e.state = "approach" if e.kind == "grub" else "idle"
                e.knock_left = 0.0
        self.log("down")

    def _set_invulnerable(self, value, why):
        if self.invulnerable != value:
            self.invulnerable = value
            self.log("invulnerable", value=value, why=why)

    def _move(self, direction, distance):
        self.x, self.y = self._clamp(
            self.x + direction[0] * distance, self.y + direction[1] * distance
        )

    @staticmethod
    def _clamp(x, y):
        return min(max(x, 0.0), ROOM_W - 1.0), min(max(y, 0.0), ROOM_H - 1.0)


def _name(direction):
    return next(k for k, v in DIRECTIONS.items() if v == direction)

usage: python3 play.py [--color] --view player|trace SCRIPT

`player` prints what a player sees and hears; `trace` prints the simulation state for developers.

ANSI = {
    "lantern": "\x1b[38;5;229m",  # warm yellow-white

    "tint": "\x1b[38;5;109m",  # muted, tinted

def describe(kind, d):

        return [LINES["run start"]]

def read_script(text):
    setup = {"room": "1", "fps": 60, "end": None, "spawns": [], "player": None}
    inputs = []
    for number, raw in enumerate(text.splitlines(), 1):
        line = raw.split("#")[0].strip()
        if not line:
            continue
        words = line.split()
        try:
            if words[0] == "room":
                setup["room"] = words[1]
            elif words[0] == "fps":
                setup["fps"] = int(words[1])
            elif words[0] == "end":
                setup["end"] = float(words[1])
            elif words[0] == "spawn":
                setup["spawns"].append((words[1], float(words[2]), float(words[3])))
            elif words[0] == "player":
                setup["player"] = (float(words[1]), float(words[2]))
            else:
                inputs.append(
                    (float(words[0]), words[1], words[2] if len(words) > 2 else None)
                )
        except (IndexError, ValueError):
            sys.exit(f"script line {number}: cannot read {raw.strip()!r}")
    return setup, sorted(inputs, key=lambda i: i[0])


def play(setup, inputs, view, color=False):
    try:
        game = Game(fps=setup["fps"], room=setup["room"])
        for kind, x, y in setup["spawns"]:
            game.spawn(kind, x, y)
        if setup["player"]:
            game.place_player(*setup["player"])
    except ValueError as error:
        sys.exit(f"script setup: {error}")
    end = setup["end"] if setup["end"] is not None else (inputs[-1][0] + 3 if inputs else 5)
    end = min(end, MAX_SECONDS)
    out = Player(game, color) if view == "player" else Trace(game)
    out.start()
    pending = list(inputs)
    while game.t < end - 1e-9:
        while pending and pending[0][0] <= game.t + 1e-9:
            when, action, direction = pending.pop(0)
            try:
                game.press(action, direction)
            except ValueError as error:
                sys.exit(f"input at {when}: {error}")
        game.step()
        out.frame()
        # The session stops at the run-over screen (unless later input could restart the run)
        # and when a practice-room session ends.
        if game.phase == "ended" or (game.phase == "over" and not pending):
            break
    out.finish()


class Player:
    """What a player sees and hears: the room, the hearts, and feedback on their actions."""

    def __init__(self, game, color=False):
        self.game, self.color, self.seen, self.next_snapshot = game, color, 0, 0.0

    def start(self):
        g = self.game
        print(f"Lantern Depths {VERSION} — room {g.room} — {g.fps} FPS")
        self.snapshot()
        self.next_snapshot = FRAME_EVERY

    def frame(self):
        g = self.game
        lines, draw = [], False
        for when, kind, details in g.events[self.seen :]:
            for text in describe(kind, details):
                lines.append((when, text, kind))
            draw = draw or kind in DRAW_ON
        self.seen = len(g.events)
        counts = {}
        for when, text, kind in lines:
            if kind in DEDUPE:
                counts[(when, text)] = counts.get((when, text), 0) + 1
        ordinal = {}
        for when, text, kind in lines:
            if counts.get((when, text), 0) > 1 and kind in DEDUPE:
                ordinal[(when, text)] = ordinal.get((when, text), 0) + 1
                text = f"{text} #{ordinal[(when, text)]}"
            print(f"{when:7.3f}  {text}")
        due = g.t >= self.next_snapshot - 1e-9
        if (draw or due) and g.phase not in ("over", "ended"):
            self.snapshot()
        if due:
            self.next_snapshot += FRAME_EVERY

    def finish(self):
        g = self.game
        left = sum(1 for e in g.enemies if e.hp > 0)
        print(f"end at {g.t:.2f} s: hearts {g.hearts}/{HEARTS}, enemies left {left}")

    describe = staticmethod(describe)

    def paint(self, glyph):
        char, color = glyph
        if self.color and color:
            return f"{ANSI[color]}{char}{RESET}"
        return char

    def snapshot(self):
        g = self.game
        grid = [[(" ", None) for _ in range(ROOM_W)] for _ in range(ROOM_H)]

        def put(x, y, glyph):
            col, row = int(round(x)), int(round(y))
            if 0 <= col < ROOM_W and 0 <= row < ROOM_H:
                grid[row][col] = glyph

        if g.arc_left > 1e-9:
            for col, row in g.arc_tiles():
                grid[row][col] = (ARCS[g.arc_dir], "lantern")
        for enemy in g.enemies:
            if enemy.hp > 0:
                put(enemy.x, enemy.y, enemy_glyph(enemy))
        for shot in g.shots:
            put(shot[0], shot[1], GLYPHS["shot"])
        put(g.x, g.y, GLYPHS["player"] if g.hearts > 0 else GLYPHS["down"])
        print(f"   [{g.t:.2f} s] hearts {g.hearts}/{HEARTS}")
        print("   +" + "-" * ROOM_W + "+")
        for row in grid:
            print("   |" + "".join(self.paint(cell) for cell in row) + "|")
        print("   +" + "-" * ROOM_W + "+")


class Trace:
    """Simulation state for developers: one line per frame in which anything happened."""

    def __init__(self, game):
        self.game, self.seen = game, 0

    def start(self):
        g = self.game
        print(f"trace: fps={g.fps} room={g.room} dt={g.dt:.5f}")
        self.line([("setup", {})])

    def frame(self):
        g = self.game
        new = [(kind, details) for _, kind, details in g.events[self.seen :]]
        self.seen = len(g.events)

"""Balance bots: let simple player strategies play rooms and report how they fared.

usage: python3 bots.py SCRIPT

Script lines (all optional; `#` starts a comment):
  bots camper,walker,dodger   which bots play (default: all)
  rooms 1-5                   which rooms, as a range or a list like 1,3,5 (default: 1-5)
  fps 60                      30 or 60 (default: 60)
  limit 60                    seconds a bot may take per room (default: 60)

Each bot starts every room with full hearts.
"""

import math
import sys

from game import ATTACK_REACH, DIRECTIONS, HEARTS, ROOMS, Game


def _toward(dx, dy):
    if abs(dx) >= abs(dy):
        return "right" if dx > 0 else "left"
    return "down" if dy > 0 else "up"


def _nearest(game):
    live = [e for e in game.enemies if e.hp > 0]
    return min(live, key=lambda e: math.hypot(e.x - game.x, e.y - game.y), default=None)


def _face_and_swing(game, enemy):
    game.press("move", _toward(enemy.x - game.x, enemy.y - game.y))
    game.press("stop")
    if math.hypot(enemy.x - game.x, enemy.y - game.y) <= ATTACK_REACH and game.recovery <= 0:
        game.press("attack")


def _incoming(game):
    """A shot that will reach the player within 0.7 s, if any."""
    for x, y, vx, vy in game.shots:
        speed = math.hypot(vx, vy)
        ahead = ((game.x - x) * vx + (game.y - y) * vy) / speed
        beside = abs((game.x - x) * vy - (game.y - y) * vx) / speed
        if 0 < ahead < 0.7 * speed and beside < 0.7:
            return vx, vy
    return None


def _sidestep(game, shot):
    vx, vy = shot
    if abs(vx) >= abs(vy):
        return "down" if game.y < 4 else "up"
    return "right" if game.x < 8 else "left"


def camper(game):
    """Holds the starting spot: turns to the nearest enemy and swings when it is in reach."""
    enemy = _nearest(game)
    if enemy:
        _face_and_swing(game, enemy)


def walker(game):
    """Lets a nearby enemy come and swings, walks up to one further away, steps aside from shots."""
    enemy = _nearest(game)
    if not enemy:
        return
    shot = _incoming(game)
    distance = math.hypot(enemy.x - game.x, enemy.y - game.y)
    coming = enemy.kind == "grub" and distance < 2.5
    if shot:
        game.press("move", _sidestep(game, shot))
    elif distance > 1.1 and not coming:
        game.press("move", _toward(enemy.x - game.x, enemy.y - game.y))
    else:
        _face_and_swing(game, enemy)


def dodger(game):
    """Plays like the walker, but dashes aside from incoming shots and away from two close enemies."""
    ready = game.dash_left <= 0 and game.dash_cooldown <= 0
    shot = _incoming(game)
    close = [e for e in game.enemies if e.hp > 0 and math.hypot(e.x - game.x, e.y - game.y) < 1.3]
    if ready and shot:
        game.press("dash", _sidestep(game, shot))
    elif ready and len(close) >= 2:
        game.press("dash", _toward(game.x - close[0].x, game.y - close[0].y))
    else:
        walker(game)


BOTS = {"camper": camper, "walker": walker, "dodger": dodger}


def play_room(bot, room, fps, limit):
    game = Game(fps=fps, room=room)
    while game.t < limit - 1e-9 and not game.cleared and game.hearts > 0:
        if game.stun_left <= 0:
            bot(game)
        game.step()
    counts = {}
    sources = {}
    for _, kind, details in game.events:
        counts[kind] = counts.get(kind, 0) + 1
        if kind == "hurt":
            sources[details["source"]] = sources.get(details["source"], 0) + 1
    return {
        "cleared": game.cleared,
        "seconds": round(game.t, 2),
        "hearts_lost": counts.get("hurt", 0),  # counted from events: a clear heals one heart
        "hits_by": sources,
        "attacks": counts.get("attack", 0),
        "dashes": counts.get("dash", 0),
    }


def read_script(text):
    setup = {"bots": list(BOTS), "rooms": [r for r in ROOMS if r != "empty"], "fps": 60, "limit": 60.0}
    for number, raw in enumerate(text.splitlines(), 1):
        line = raw.split("#")[0].strip()
        if not line:
            continue
        key, _, value = line.partition(" ")
        value = value.replace(" ", "")
        try:
            if key == "bots":
                setup["bots"] = value.split(",")
                assert all(b in BOTS for b in setup["bots"])
            elif key == "rooms":
                if "-" in value:
                    first, last = value.split("-")
                    value = ",".join(str(n) for n in range(int(first), int(last) + 1))
                setup["rooms"] = value.split(",")
                assert all(r in ROOMS for r in setup["rooms"])
            elif key == "fps":
                setup["fps"] = int(value)
            elif key == "limit":
                setup["limit"] = float(value)
            else:
                raise ValueError
        except (AssertionError, ValueError):
            sys.exit(f"script line {number}: cannot read {raw.strip()!r}")
    return setup


def main():
    with open(sys.argv[1], encoding="utf-8") as f:
        setup = read_script(f.read())
    print(f"bots at {setup['fps']} FPS, up to {setup['limit']:.0f} s per room, {HEARTS} hearts each")
    for name in setup["bots"]:
        total = {"cleared": 0, "hearts_lost": 0, "dashes": 0}
        for room in setup["rooms"]:
            r = play_room(BOTS[name], room, setup["fps"], setup["limit"])
            hits = ", ".join(f"{k} {v}" for k, v in sorted(r["hits_by"].items())) or "none"
            outcome = f"cleared in {r['seconds']:.1f} s" if r["cleared"] else f"NOT cleared ({r['seconds']:.1f} s)"
            print(
                f"{name:7} room {room}: {outcome}; hearts lost {r['hearts_lost']} (hit by: {hits}); "
                f"attacks {r['attacks']}, dashes {r['dashes']}"
            )
            total["cleared"] += r["cleared"]
            total["hearts_lost"] += r["hearts_lost"]
            total["dashes"] += r["dashes"]
        print(
            f"{name:7} total: cleared {total['cleared']}/{len(setup['rooms'])}, hearts lost "
            f"{total['hearts_lost']}, dashes {total['dashes']}"
        )


if __name__ == "__main__":
    main()

class SpitterTest(unittest.TestCase):
    def test_stagger_schedule(self):  # AC-7

                        self.assertAlmostEqual(t, 1.0 + 0.8 * k + 2.5 * n, delta=tol(fps), msg=(fps, room, k))
                for kind in ("charge", "spit"):
                    ts = sorted(times(game, kind))
                    self.assertTrue(all(b - a >= 0.5 for a, b in zip(ts, ts[1:])), (fps, room, kind, ts))

    def test_spit_lines_name_their_spitter(self):  # AC-8, defect 15
        for fps in (30, 60):
            for room in ("3", "4", "5"):
                setup = {"room": room, "fps": fps, "end": 9.0, "spawns": [], "player": None}
                text = capture(setup, [], "player")
                rows = [m.groups() for m in (re.match(r"\s*(\d+\.\d{3})  (.+)$", l) for l in text.splitlines()) if m]
                spits = [(float(t), re.match(r"spitter \((.+)\) spits$", s).group(1)) for t, s in rows if re.match(r"spitter \(.+\) spits$", s)]
                charges = [(float(t), re.match(r"spitter \((.+)\) charges$", s).group(1)) for t, s in rows if re.match(r"spitter \(.+\) charges$", s)]
                self.assertTrue(spits, (fps, room))
                texts = [(t, s) for t, s in rows]
                self.assertEqual(len(texts), len(set(texts)), (fps, room))  # no identical line at one timestamp
                for t, who in spits:
                    self.assertTrue(
                        any(w == who and abs((t - tc) - 0.8) <= tol(fps) for tc, w in charges), (fps, room, t, who)
                    )
                trace = capture(setup, [], "trace")
                self.assertEqual(len(re.findall(r"\bspit \{", trace)), len(spits), (fps, room))
