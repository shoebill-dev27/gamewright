class VariantMessageTest(unittest.TestCase):
    """D-011: wind-up and hit messages name the variant; everything else is unchanged."""

    def trace_session(self, script):
        import contextlib
        import io

        import play

        setup, inputs, rules = play.read_script(script)
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            play.play(setup, inputs, "trace", rules)
        return out.getvalue()

    def test_wind_up_message_names_the_variant_once(self):
        import play

        for fps in (30, 60):
            for kind in ("grub", "runner", "brute"):
                game = Game(fps=fps, room="empty")
                game.spawn(kind, 10, 4)
                run(game, 0.6)
                texts = [play.Player.describe(k, d) for _, k, d in game.events if k == "windup"]
                self.assertEqual(texts, [f"a {kind} rears back"], (fps, kind))

    def test_hit_message_names_the_variant_and_keeps_the_hearts_text(self):
        import play

        for fps in (30, 60):
            for kind in ("grub", "runner", "brute"):
                game = Game(fps=fps, room="empty")
                game.spawn(kind, 8.5, 4)
                self.assertTrue(step_until(game, has_event("hurt")), (fps, kind))
                texts = [play.Player.describe(k, d) for _, k, d in game.events if k == "hurt"]
                self.assertEqual(texts, [f"YOU ARE HIT by a {kind} — hearts 2/3"], (fps, kind))

    def test_shot_hit_message_is_unchanged(self):
        import play

        game = Game(room="empty")
        game.shots.append([game.x + 0.2, game.y, 0.0, 0.0, 0.0])
        game.step()
        texts = [play.Player.describe(k, d) for _, k, d in game.events if k == "hurt"]
        self.assertEqual(texts, ["YOU ARE HIT by a shot — hearts 2/3"])

    def test_rears_back_rule_reacts_to_all_variants(self):
        for kind in ("grub", "runner", "brute"):
            out = ReactionTest.session(
                self, f'room empty\nspawn {kind} 10 4\nwhen text "rears back" then dash up\nend 1.5\n'
            )
            self.assertIn("you dash up", out, kind)

    def test_old_grub_rule_matches_only_the_standard_grub(self):
        for kind, reacts in (("grub", True), ("runner", False), ("brute", False)):
            out = ReactionTest.session(
                self,
                f'room empty\nspawn {kind} 10 4\nwhen text "a grub rears back" then dash up\nend 1.5\n',
            )
            self.assertEqual("you dash up" in out, reacts, kind)

    def test_trace_hurt_events_are_unchanged(self):
        out = self.trace_session("room empty\nspawn runner 8.5 4\nend 2\n")
        hurt = [line for line in out.splitlines() if "hurt {" in line]
        self.assertTrue(hurt)
        for line in hurt:
            self.assertNotIn("'enemy'", line)
