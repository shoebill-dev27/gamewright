class VariantMessageTest(unittest.TestCase):
    """D-011: wind-up and hit messages name the variant; everything else is unchanged."""

    KINDS = ("grub", "runner", "brute")

    def session(self, script, view="player"):
        import contextlib
        import io

        import play

        setup, inputs, rules = play.read_script(script)
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            play.play(setup, inputs, view, rules)
        return out.getvalue()

    def script(self, kind, fps, x, extra=""):
        return f"room empty\nfps {fps}\nspawn {kind} {x} 4\nplayer 8 4\n{extra}end 2.5\n"

    def test_each_wind_up_prints_its_own_message_once(self):
        for fps in (30, 60):
            for kind in self.KINDS:
                out = self.session(self.script(kind, fps, 10))
                self.assertEqual(out.count(f"a {kind} rears back"), 1, (fps, kind))
                for other in self.KINDS:
                    if other != kind:
                        self.assertNotIn(f"a {other} rears back", out, (fps, kind))

    def test_wind_up_message_is_at_the_wind_up_moment(self):
        for fps in (30, 60):
            for kind in self.KINDS:
                game = Game(fps=fps, room="empty")
                game.spawn(kind, 10, 4)
                self.assertTrue(step_until(game, has_event("windup")))
                when = times(game, "windup")[0]
                out = self.session(self.script(kind, fps, 10))
                self.assertIn(f"{when:7.3f}  a {kind} rears back", out, (fps, kind))

    def test_each_hit_prints_the_matching_message_on_the_frame_of_contact(self):
        for fps in (30, 60):
            for kind in self.KINDS:
                game = Game(fps=fps, room="empty")
                game.spawn(kind, 8.5, 4)
                game.place_player(8, 4)
                self.assertTrue(step_until(game, has_event("hurt"), 3.0))
                when = times(game, "hurt")[0]
                out = self.session(self.script(kind, fps, 8.5))
                self.assertIn(
                    f"{when:7.3f}  YOU ARE HIT by a {kind} — hearts 2/3", out, (fps, kind)
                )

    def test_shot_hit_and_other_messages_are_unchanged(self):
        import play

        d = play.Player.describe
        self.assertEqual(d("hurt", {"source": "shot", "hearts": 2}), "YOU ARE HIT by a shot — hearts 2/3")
        self.assertEqual(d("hurt", {"source": "grub", "hearts": 1}), "YOU ARE HIT by a grub — hearts 1/3")
        self.assertEqual(d("aim", {}), "a spitter takes aim")
        self.assertEqual(d("spit", {}), "a spitter spits")
        self.assertEqual(d("shrug", {"enemy": "brute"}), "the brute shrugs off the swipe")

    def test_rears_back_rule_reacts_to_all_variants(self):
        for fps in (30, 60):
            for kind in self.KINDS:
                out = self.session(
                    self.script(kind, fps, 10, 'when text "rears back" then dash away\n')
                )
                self.assertIn("you dash", out, (fps, kind))

    def test_old_keyed_rules_match_only_the_standard_grub(self):
        for kind, expect in (("grub", True), ("runner", False), ("brute", False)):
            out = self.session(
                self.script(kind, 60, 10, 'when text "a grub rears back" then dash away\n')
            )
            self.assertEqual("you dash" in out, expect, kind)

    def test_trace_and_bot_sources_are_unchanged(self):
        out = self.session(self.script("brute", 60, 8.5), view="trace")
        hurt = [l for l in out.splitlines() if "hurt {" in l]
        self.assertTrue(hurt)
        for line in hurt:
            self.assertIn("hurt {'source': 'grub', 'hearts': 2}", line)
            self.assertNotIn("'enemy'", line)
        game = Game(room="empty")
        game.spawn("runner", 8.5, 4)
        self.assertTrue(step_until(game, has_event("hurt"), 3.0))
        details = next(d for _, k, d in game.events if k == "hurt")
        self.assertEqual(details["source"], "grub")
        self.assertEqual(details["enemy"], "runner")


if __name__ == "__main__":
