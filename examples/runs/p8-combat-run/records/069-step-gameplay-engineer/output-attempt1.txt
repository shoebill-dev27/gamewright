class VariantMessageTest(unittest.TestCase):
    """D-011: wind-up and hit messages name the variant (V-a to V-d)."""

    NAMES = (("grub", "grub"), ("runner", "runner"), ("brute", "brute"))

    def session(self, script):
        return ReactionTest.session(self, script)

    def test_windup_and_hit_name_the_variant_once_each(self):
        for fps in (30, 60):
            for kind, noun in self.NAMES:
                out = self.session(
                    f"room empty\nfps {fps}\nspawn {kind} 9 4\nplayer 8 4\nend 0.9\n"
                )
                self.assertEqual(out.count(f"a {noun} rears back"), 1, (fps, kind))
                self.assertEqual(out.count(f"YOU ARE HIT by a {noun} — hearts 2/3"), 1, (fps, kind))
                for other, other_noun in self.NAMES:
                    if other != kind:
                        self.assertNotIn(f"a {other_noun} rears back", out, (fps, kind))
                        self.assertNotIn(f"by a {other_noun}", out, (fps, kind))

    def test_the_shot_message_is_unchanged(self):
        import play

        self.assertEqual(
            play.Player.describe("hurt", {"source": "shot", "hearts": 2}),
            "YOU ARE HIT by a shot — hearts 2/3",
        )

    def test_a_rule_on_rears_back_reacts_to_every_variant(self):
        for fps in (30, 60):
            for kind, _ in self.NAMES:
                out = self.session(
                    f'room empty\nfps {fps}\nspawn {kind} 10 4\nplayer 8 4\n'
                    'when text "rears back" then dash up\nend 1.5\n'
                )
                self.assertIn("you dash up", out, (fps, kind))

    def test_the_trace_does_not_show_the_variant_detail(self):
        import contextlib

        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            play.play(setup, inputs, "trace", rules)
        self.assertIn("hurt {'source': 'grub', 'hearts': 2}", out.getvalue())
        self.assertNotIn("variant", out.getvalue())
