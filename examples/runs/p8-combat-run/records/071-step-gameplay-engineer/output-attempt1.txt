class VariantMessageTest(unittest.TestCase):
    """D-011: wind-up and hit messages name the variant (V-a to V-d)."""

    NOUNS = {"grub": "grub", "runner": "runner", "brute": "brute"}

    def session(self, script, view="player"):
        import contextlib
        import io

        import play

        setup, inputs, rules = play.read_script(script)
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            play.play(setup, inputs, view, rules)
        return out.getvalue()

    def test_each_wind_up_prints_its_own_message_once(self):
        for fps in (30, 60):
            for kind, noun in self.NOUNS.items():
                out = self.session(f"room empty\nfps {fps}\nspawn {kind} 10 4\nend 1.0\n")
                self.assertEqual(out.count("rears back"), 1, (fps, kind))
                self.assertEqual(out.count(f"a {noun} rears back"), 1, (fps, kind))

    def test_each_hit_prints_the_matching_message_with_the_same_hearts_text(self):
        for fps in (30, 60):
            for kind, noun in self.NOUNS.items():
                out = self.session(f"room empty\nfps {fps}\nspawn {kind} 8.5 4\nend 2\n")
                self.assertEqual(out.count("YOU ARE HIT"), 1, (fps, kind))
                self.assertIn(f"YOU ARE HIT by a {noun} — hearts 2/3", out, (fps, kind))

    def test_a_shot_hit_still_says_shot(self):
        out = self.session("room empty\nspawn spitter 1 4\nend 5\n")
        self.assertIn("YOU ARE HIT by a shot — hearts 2/3", out)

    def test_a_rears_back_rule_reacts_to_every_variant(self):
        for kind in self.NOUNS:
            out = self.session(
                f'room empty\nspawn {kind} 10 4\nwhen text "rears back" then dash up\nend 1.5\n'
            )
            self.assertIn("you dash up", out, kind)
            self.assertNotIn("YOU ARE HIT", out, kind)

    def test_a_rule_keyed_on_a_grub_matches_only_the_standard_grub(self):
        for kind, reacts in (("grub", True), ("runner", False), ("brute", False)):
            out = self.session(
                f'room empty\nspawn {kind} 10 4\nwhen text "a grub rears back" then dash up\nend 1.5\n'
            )
            self.assertEqual("you dash up" in out, reacts, kind)
