class Room3Test(unittest.TestCase):
    """D-012 / R3-1, R3-6, R3-7, R3-8: room 3 has three spitters and one standard grub."""

    def test_the_grub_changes_no_cue_time_for_a_still_player(self):
        for fps in (30, 60):
            with_grub = Game(fps=fps, room="3")
            without = Game(fps=fps, room="3")
            without.enemies = [e for e in without.enemies if e.kind == "spitter"]
            for g in (with_grub, without):
                g.hearts = 50
                run(g, 6.0)  # before the grub can reach the player
            self.assertEqual(times(with_grub, "aim"), times(without, "aim"), fps)
            self.assertEqual(times(with_grub, "spit"), times(without, "spit"), fps)

    def test_the_grub_does_not_wind_up_before_4_s_for_a_still_player(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="3")
            game.hearts = 50
            run(game, 4.0)
            self.assertEqual(times(game, "windup"), [], fps)

    def test_the_room_clears_only_when_spitters_and_grub_are_all_dead(self):
        for fps in (30, 60):
            game = Game(fps=fps, room="3")
            spitters = [e for e in game.enemies if e.kind == "spitter"]
            grub = [e for e in game.enemies if e.is_grub]
            self.assertEqual((len(spitters), len(grub)), (3, 1), fps)
            for e in spitters:
                e.hp = 0
            run(game, 0.2)
            self.assertFalse(game.cleared, fps)
            grub[0].hp = 0
            run(game, 0.1)
            self.assertTrue(game.cleared, fps)
            game = Game(fps=fps, room="3")
            game.enemies[3].hp = 0
            run(game, 0.2)
            self.assertFalse(game.cleared, fps)

    def test_no_two_spitters_are_more_than_9_5_tiles_apart(self):
        import math

        pos = [(x, y) for kind, x, y in ROOMS["3"] if kind == "spitter"]
        for a in pos:
            for b in pos:
                self.assertLessEqual(math.hypot(a[0] - b[0], a[1] - b[1]), 9.5)
